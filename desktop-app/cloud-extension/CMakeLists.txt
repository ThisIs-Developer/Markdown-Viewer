cmake_minimum_required(VERSION 3.16)
project(MarkdownViewerCloudExtension LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This first adapter preserves the existing Windows credential backend. Other platforms need a keychain adapter.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(markdown-cloud WIN32 main.cpp providers.cpp credentials.cpp webdav-credentials.cpp)
target_compile_features(markdown-cloud PRIVATE cxx_std_17)
target_compile_definitions(markdown-cloud PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(markdown-cloud PRIVATE winhttp advapi32)
if(MSVC)
  target_compile_options(markdown-cloud PRIVATE /W4 /EHsc /utf-8)
  target_link_options(markdown-cloud PRIVATE /ENTRY:mainCRTStartup)
endif()
